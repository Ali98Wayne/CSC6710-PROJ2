CREATE DATABASE proj_2;
USE proj_2;

CREATE TABLE Users(
   user_id INT AUTO_INCREMENT PRIMARY KEY,
   username VARCHAR(50),
   password VARCHAR(60),
   first_name VARCHAR(50),
   last_name VARCHAR(50),
   address_street VARCHAR(50),
   address_city VARCHAR(50),
   address_state VARCHAR(2),
   address_zip VARCHAR(5),
   phone VARCHAR(12),
   email VARCHAR(50),
   credit_card_num VARCHAR(50),
   credit_card_month VARCHAR(20),
   credit_card_year VARCHAR(4),
   credit_card_cvv VARCHAR(4),
   signup_date DATE,
   last_login DATETIME
);

CREATE TABLE Request_Cleaning(
   request_id INT AUTO_INCREMENT PRIMARY KEY,
   client_id INT NOT NULL,
   service_address_street VARCHAR(50),
   service_address_city VARCHAR(50),
   service_address_state VARCHAR(2),
   service_address_zip VARCHAR(5),
   cleaning_type ENUM('Basic', 'Deep Cleaning', 'Move-Out') NOT NULL,
   rooms INT(20),
   preferred_date DATETIME,
   proposed_budget FLOAT,
   notes TEXT NULL,
   photo_urls JSON NULL,
   request_date DATE DEFAULT CURRENT_DATE,
   quote_accept_date DATE,
   order_generated TINYINT(1) DEFAULT 0,
   bill_generated TINYINT(1) DEFAULT 0,
   status ENUM('pending','quoted','rejected') DEFAULT 'pending',
   anna_note TEXT,
   FOREIGN KEY (client_id) REFERENCES Users(user_id)
);

CREATE TABLE Quotes (
    quote_id INT AUTO_INCREMENT PRIMARY KEY,
    request_id INT NOT NULL,
    client_id INT NOT NULL,
    responder_type ENUM('Anna','Client') NOT NULL DEFAULT 'Anna',
    quote_price DECIMAL(10,2) NULL,
    scheduled_start DATETIME NULL,
    scheduled_end DATETIME NULL,
    note TEXT,
    response_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status ENUM('pending','accepted','rejected','countered','canceled') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_request FOREIGN KEY (request_id) REFERENCES Request_Cleaning(request_id),
    CONSTRAINT fk_responder FOREIGN KEY (client_id) REFERENCES Users(user_id)
);

CREATE TABLE Bills (
    bill_id INT AUTO_INCREMENT PRIMARY KEY,
    request_id INT NOT NULL,        -- Links to the service request
    client_id INT NOT NULL,         -- Who is being billed
    bill_amount DECIMAL(10,2) NOT NULL,
    bill_status ENUM('Paid', 'Unpaid', 'Disputed') DEFAULT 'Unpaid',
    due_date DATE NOT NULL,
    payment_date DATE,              -- Filled when client pays
    note TEXT,                      -- Any additional note (dispute, explanation)
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_bill_request FOREIGN KEY (request_id) REFERENCES Request_Cleaning(request_id),
    CONSTRAINT fk_bill_client FOREIGN KEY (client_id) REFERENCES Users(user_id)
);

CREATE TABLE Quote_History (
    history_id INT AUTO_INCREMENT PRIMARY KEY,
    quote_id INT NOT NULL,
    client_id INT NOT NULL,
    responder_type ENUM('Anna','Client') NOT NULL,
    quote_price DECIMAL(10,2) NULL,
    scheduled_start DATETIME NULL,
    scheduled_end DATETIME NULL,
    status ENUM('pending','accepted','rejected','countered','canceled'),
    note TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_quote_id FOREIGN KEY (quote_id) REFERENCES Quotes(quote_id),
    CONSTRAINT fk_quote_client FOREIGN KEY (client_id) REFERENCES Users(user_id)
);

CREATE TABLE Bill_History (
    history_id INT AUTO_INCREMENT PRIMARY KEY,
    bill_id INT NOT NULL,
    client_id INT NOT NULL,
    responder_type ENUM('Anna','Client') NOT NULL,
    note TEXT,
    new_amount DECIMAL(10,2),
    bill_status ENUM('Paid', 'Unpaid', 'Disputed'),
    due_date DATE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_bill_history_bill FOREIGN KEY (bill_id) REFERENCES Bills(bill_id),
    CONSTRAINT fk_bill_history_responder FOREIGN KEY (client_id) REFERENCES Users(user_id)
);

-- Most service orders query
SELECT Users.user_id AS client_id, Users.username, Users.first_name, Users.last_name, COUNT(Request_Cleaning.request_id) AS total_requests
FROM Request_Cleaning JOIN Users ON Request_Cleaning.client_id = Users.user_id
GROUP BY Users.user_id
ORDER BY total_requests DESC
LIMIT 10;

-- Accepted quotes in a given month query
SELECT r.request_id, u.user_id AS client_id, u.username, u.first_name, u.last_name, r.quote_accept_date 
FROM Request_Cleaning r JOIN Users u ON r.client_id = u.user_id
WHERE MONTH(r.quote_accept_date) = 12 -- For the month of December 
ORDER BY r.quote_accept_date DESC;

-- Largest job query
SELECT r.request_id, u.user_id AS client_id, u.username, u.first_name, u.last_name, r.rooms 
FROM Request_Cleaning r JOIN Users u ON r.client_id = u.user_id
ORDER BY r.rooms DESC
LIMIT 10;

-- Bad clients query
SELECT DISTINCT u.user_id AS client_id, u.username, u.first_name, u.last_name
FROM Users u JOIN Bills b ON u.user_id = b.client_id
WHERE b.bill_status = 'Unpaid' AND b.due_date < (CURDATE() - INTERVAL 7 DAY);

-- Uncommitted clients query
SELECT u.user_id, u.first_name, u.last_name, u.email, COUNT(r.request_id) AS request_count
FROM Users u JOIN Request_Cleaning r ON u.user_id = r.client_id
WHERE r.order_generated = 0
GROUP BY u.user_id 
HAVING request_count >= 3

-- Prospective clients query
SELECT u.user_id, u.first_name, u.last_name, u.email 
FROM Users u LEFT JOIN Request_Cleaning r ON u.user_id = r.client_id
WHERE r.client_id IS NULL

-- Overdue bills query
SELECT * FROM Bills
WHERE bill_status = 'Unpaid' AND due_date < NOW() - INTERVAL 7 DAY

-- Good clients query
SELECT u.user_id, u.first_name, u.last_name, u.email
FROM Users u JOIN Bills b ON u.user_id = b.client_id
GROUP BY u.user_id
HAVING SUM(TIMESTAMPDIFF(HOUR, b.created_at, b.payment_date) > 24) = 0
